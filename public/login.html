// Login.js – fixes hidden button issue const emailInput = document.getElementById('email'); const passwordInput = document.getElementById('password'); const loginBtn = document.getElementById('loginBtn'); const errorDiv = document.createElement('div'); // create error display if missing errorDiv.className = 'error-message'; errorDiv.style.cssText = 'color:#ff4c4c; margin-top:10px; font-size:14px;'; const loginForm = document.querySelector('form'); // Insert error div after form if not present if (!document.getElementById('login-error')) { loginForm.parentNode.insertBefore(errorDiv, loginForm.nextSibling); errorDiv.id = 'login-error'; } function isValidEmail(email) { return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email); } function toggleButton() { if (emailInput.value && passwordInput.value && isValidEmail(emailInput.value)) { loginBtn.classList.remove('hidden'); // SHOW the button loginBtn.disabled = false; } else { loginBtn.classList.add('hidden'); // HIDE the button loginBtn.disabled = true; } } // Attach input events emailInput.addEventListener('input', toggleButton); passwordInput.addEventListener('input', toggleButton); toggleButton(); // initial check // Override form submission with fetch (JSON) loginForm.addEventListener('submit', async (e) => { e.preventDefault(); const errorDiv = document.getElementById('login-error'); errorDiv.textContent = ''; const email = emailInput.value.trim(); const password = passwordInput.value; if (!email || !password) { errorDiv.textContent = 'Please fill all fields.'; return; } if (!isValidEmail(email)) { errorDiv.textContent = 'Invalid email format.'; return; } loginBtn.disabled = true; loginBtn.textContent = 'Logging in...'; try { const csrfToken = document.querySelector('meta[name="csrf-token"]')?.getAttribute('content') || ''; const res = await fetch('/login', { method: 'POST', headers: { 'Content-Type': 'application/json', ...(csrfToken ? { 'X-CSRF-Token': csrfToken } : {}) }, credentials: 'include', body: JSON.stringify({ email, password }) }); const data = await res.json(); if (res.ok && data.success) { window.location.href = data.redirect || '/index.html'; } else { errorDiv.textContent = data.error || 'Login failed. Please try again.'; loginBtn.disabled = false; loginBtn.textContent = 'Login'; loginBtn.classList.remove('hidden'); } } catch (err) { console.error('Login error:', err); errorDiv.textContent = 'Network error. Please try again.'; loginBtn.disabled = false; loginBtn.textContent = 'Login'; loginBtn.classList.remove('hidden'); } }); // If already logged in, redirect (async function checkSession() { try { const res = await fetch('/api/session', { credentials: 'include' }); const session = await res.json(); if (session.loggedIn) { window.location.href = '/index.html'; } } catch(e) {} })();
